class Vehicle {
    int getSpeed() {
        return 0;
    }
}
class Bike extends Vehicle {
    int getSpeed() {
        return 60;
    }
}
class Car extends Vehicle {
    int getSpeed() {
        return 120;
    }
}
class Bus extends Vehicle {
    int getSpeed() {
        return 80;
    }
}
public class Main {
    public static void main(String[] args) {

        Bike b = new Bike();
        Car c = new Car();
        Bus bus = new Bus();

        System.out.println("Bike Speed: " + b.getSpeed());
        System.out.println("Car Speed: " + c.getSpeed());
        System.out.println("Bus Speed: " + bus.getSpeed());
    }
}
